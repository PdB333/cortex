# Platform-neutral artifact-reader build. No Windows runtime, graphics,
# debugger, injection or external analysis engine is linked here.
cmake_minimum_required(VERSION 3.21)
project(cortex_static LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
include(FetchContent)
set(FETCHCONTENT_UPDATES_DISCONNECTED ON)
option(CORTEX_STATIC_SANITIZERS "Instrument portable parsers/tests with ASan+UBSan" OFF)
option(CORTEX_STATIC_FUZZ "Build coverage-guided libFuzzer targets (Clang)" OFF)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git
    GIT_TAG 9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03)
set(JSON_BuildTests OFF CACHE INTERNAL "")
FetchContent_MakeAvailable(json)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SOURCES
    artifact.cpp registry_default.cpp pe_provider.cpp pe_extras.cpp
    asar_provider.cpp js_lexer.cpp js_analysis.cpp js_provider.cpp source_map.cpp
    inflate.cpp zip_provider.cpp appx_metadata.cpp
    dotnet_metadata.cpp cil.cpp dotnet_provider.cpp
    minidump_provider.cpp hive_provider.cpp pdb_provider.cpp elf_provider.cpp
    cab_provider.cpp compound_reader.cpp compound_provider.cpp msi_tables.cpp archive_provider.cpp certificate_provider.cpp typelib_provider.cpp package_metadata.cpp
    bytecode_provider.cpp native_more_provider.cpp data_provider.cpp
    capture_provider.cpp firmware_provider.cpp document_provider.cpp)
list(TRANSFORM SOURCES PREPEND "${ROOT}/core/artifacts/")
add_library(cortex_static_parsers STATIC ${SOURCES})
target_include_directories(cortex_static_parsers PUBLIC "${ROOT}/core")
target_link_libraries(cortex_static_parsers PUBLIC nlohmann_json::nlohmann_json)
if(CORTEX_STATIC_SANITIZERS OR CORTEX_STATIC_FUZZ)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        message(FATAL_ERROR "This sanitizer configuration requires Clang or GCC")
    endif()
    target_compile_options(cortex_static_parsers PUBLIC -fsanitize=address,undefined
        -fno-sanitize-recover=undefined -fno-omit-frame-pointer -g)
    target_link_options(cortex_static_parsers PUBLIC -fsanitize=address,undefined)
endif()
if(CORTEX_STATIC_FUZZ)
    if(MINGW)
        message(FATAL_ERROR "This MinGW target does not accept libFuzzer flags. Use CORTEX_STATIC_FUZZ=OFF, ASan/UBSan, and cortex_fuzz_wave1_mutations; use native Clang for libFuzzer.")
    endif()
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "libFuzzer targets require Clang")
    endif()
    target_compile_options(cortex_static_parsers PRIVATE -fsanitize=fuzzer-no-link)
    target_link_options(cortex_static_parsers PUBLIC -fsanitize=fuzzer-no-link)
endif()
add_executable(cortex-inspect main.cpp)
target_link_libraries(cortex-inspect PRIVATE cortex_static_parsers)
if(BUILD_TESTING)
    add_executable(cortex_format_next_tests "${ROOT}/tests/format_next_tests.cpp")
    target_link_libraries(cortex_format_next_tests PRIVATE cortex_static_parsers)
    add_test(NAME cortex_format_next_tests COMMAND cortex_format_next_tests "${ROOT}/tests/fixtures")
    add_executable(cortex_fuzz_format_next_mutations "${ROOT}/tests/fuzz/fuzz_format_next.cpp" "${ROOT}/tests/fuzz/mutation_main.cpp")
    target_link_libraries(cortex_fuzz_format_next_mutations PRIVATE cortex_static_parsers)
    add_executable(cortex_fuzz_format_next_replay "${ROOT}/tests/fuzz/fuzz_format_next.cpp" "${ROOT}/tests/fuzz/replay_main.cpp")
    target_link_libraries(cortex_fuzz_format_next_replay PRIVATE cortex_static_parsers)
    add_test(NAME cortex_fuzz_format_next_regressions COMMAND cortex_fuzz_format_next_replay "${ROOT}/tests/fuzz/regressions/fuzz_format_next")
    add_executable(cortex_format_wave1_tests "${ROOT}/tests/format_wave1_tests.cpp")
    target_link_libraries(cortex_format_wave1_tests PRIVATE cortex_static_parsers)
    add_test(NAME cortex_format_wave1_tests COMMAND cortex_format_wave1_tests "${ROOT}/tests/fixtures")
    add_executable(cortex_static_dotnet_tests "${ROOT}/tests/dotnet_tests.cpp")
    target_link_libraries(cortex_static_dotnet_tests PRIVATE cortex_static_parsers)
    add_test(NAME cortex_static_dotnet_tests COMMAND cortex_static_dotnet_tests "${ROOT}/tests/fixtures/dotnet")
    add_executable(cortex_fuzz_wave1_replay "${ROOT}/tests/fuzz/fuzz_wave1.cpp" "${ROOT}/tests/fuzz/replay_main.cpp")
    target_link_libraries(cortex_fuzz_wave1_replay PRIVATE cortex_static_parsers)
    add_test(NAME cortex_fuzz_wave1_regressions COMMAND cortex_fuzz_wave1_replay "${ROOT}/tests/fuzz/regressions/fuzz_wave1")
    add_executable(cortex_fuzz_wave1_mutations "${ROOT}/tests/fuzz/fuzz_wave1.cpp" "${ROOT}/tests/fuzz/mutation_main.cpp")
    target_link_libraries(cortex_fuzz_wave1_mutations PRIVATE cortex_static_parsers)
    add_executable(cortex_fuzz_format_more_replay "${ROOT}/tests/fuzz/fuzz_format_more.cpp" "${ROOT}/tests/fuzz/replay_main.cpp")
    target_link_libraries(cortex_fuzz_format_more_replay PRIVATE cortex_static_parsers)
    add_test(NAME cortex_fuzz_format_more_regressions COMMAND cortex_fuzz_format_more_replay "${ROOT}/tests/fuzz/regressions/fuzz_format_more")
    add_executable(cortex_fuzz_format_more_mutations "${ROOT}/tests/fuzz/fuzz_format_more.cpp" "${ROOT}/tests/fuzz/mutation_main.cpp")
    target_link_libraries(cortex_fuzz_format_more_mutations PRIVATE cortex_static_parsers)
    add_executable(cortex_format_more_tests "${ROOT}/tests/format_more_tests.cpp")
    target_link_libraries(cortex_format_more_tests PRIVATE cortex_static_parsers)
    add_test(NAME cortex_format_more_tests COMMAND cortex_format_more_tests "${ROOT}/tests/fixtures")
    add_executable(cortex_static_artifact_tests "${ROOT}/tests/artifacts_tests.cpp")
    target_link_libraries(cortex_static_artifact_tests PRIVATE cortex_static_parsers)
    add_test(NAME cortex_static_artifact_tests COMMAND cortex_static_artifact_tests)
    set_tests_properties(cortex_static_artifact_tests PROPERTIES WORKING_DIRECTORY "${ROOT}")
endif()
if(CORTEX_STATIC_FUZZ)
    foreach(harness fuzz_artifact fuzz_wave1 fuzz_format_next fuzz_format_more)
        add_executable(${harness} "${ROOT}/tests/fuzz/${harness}.cpp")
        target_compile_options(${harness} PRIVATE -fsanitize=fuzzer)
        target_link_options(${harness} PRIVATE -fsanitize=fuzzer)
        target_link_libraries(${harness} PRIVATE cortex_static_parsers)
    endforeach()
endif()
