cmake_minimum_required(VERSION 3.21)
project(cortex_remote_agent LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(CORTEX_REMOTE_SANITIZERS "Instrument Cortex Remote and TLS with ASan/UBSan" OFF)
option(CORTEX_REMOTE_PS5 "Compile the native backend explicitly for the PS5 SDK" OFF)
option(CORTEX_REMOTE_FUZZ "Build the pure protocol decoder libFuzzer target" OFF)
if(CORTEX_REMOTE_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=address,undefined)
endif()
set(FETCHCONTENT_UPDATES_DISCONNECTED ON)
include(../../cmake/CortexRemote.cmake)
add_executable(cortex-remote main.cpp)
target_link_libraries(cortex-remote PRIVATE cortex_remote)
add_executable(cortex-agent ../../agent/remote/main.cpp ../../agent/remote/backend_freebsd.cpp)
if(NOT CORTEX_REMOTE_PS5)
    target_sources(cortex-agent PRIVATE ../../agent/remote/backend_fixture.cpp)
endif()
target_link_libraries(cortex-agent PRIVATE cortex_remote)
if(CORTEX_REMOTE_PS5)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "FreeBSD")
        message(FATAL_ERROR "PS5 builds require the SDK FreeBSD target, not a host executable")
    endif()
    target_compile_definitions(cortex-agent PRIVATE CORTEX_REMOTE_PS5=1)
    set_target_properties(cortex-agent PROPERTIES SUFFIX ".elf")
endif()
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    add_executable(cortex_remote_protocol_tests ../../tests/remote_protocol_tests.cpp)
    target_link_libraries(cortex_remote_protocol_tests PRIVATE cortex_remote)
    add_test(NAME remote_protocol COMMAND cortex_remote_protocol_tests)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME remote_agent_e2e COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/remote_agent_e2e.py
        --client $<TARGET_FILE:cortex-remote> --agent $<TARGET_FILE:cortex-agent>
        --work ${CMAKE_BINARY_DIR}/e2e-credentials)
    set_tests_properties(remote_agent_e2e PROPERTIES TIMEOUT 150)
endif()
if(CORTEX_REMOTE_FUZZ)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR MINGW)
        message(FATAL_ERROR "Coverage-guided fuzzing needs native Clang (not MinGW)")
    endif()
    add_executable(fuzz_remote_protocol ../../tests/fuzz/fuzz_remote_protocol.cpp ../../core/remote/protocol.cpp)
    target_link_libraries(fuzz_remote_protocol PRIVATE nlohmann_json::nlohmann_json)
    target_compile_options(fuzz_remote_protocol PRIVATE -fsanitize=fuzzer,address,undefined -fno-sanitize-recover=undefined -g)
    target_link_options(fuzz_remote_protocol PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
